const handleSelectionChange = (selection) => {
  selectedRows.value = selection

// 移动端复选框变化
const handleMobileCheckChange = (item) => {
  if (item.checked) {
    if (!selectedRows.value.find(r => r.id === item.id)) {
      selectedRows.value.push(item)
    }
  } else {
    selectedRows.value = selectedRows.value.filter(r => r.id !== item.id)
  }
}

**步骤4: 添加提交审核函数**

在 `handleEdit` 函数后添加:
```javascript
// 单条提交审核
const handleSubmit = async (row) => {
  try {
    await ElMessageBox.confirm(
      t('farmerDemand.submitConfirm'),
      t('common.tip'),
      {
        confirmButtonText: t('common.confirm'),
        cancelButtonText: t('common.cancel'),
        type: 'warning'
      }
    )
    const res = await submitForAudit([row.id])
    if (res.code === 200) {
      const result = res.data
      if (result.successCount > 0) {
        ElMessage.success(t('farmerDemand.submitSuccess'))
        loadData()
      } else {
        ElMessage.error(t('farmerDemand.submitFailed'))
      }
    } else {
      ElMessage.error(res.msg || t('farmerDemand.submitFailed'))
    }
  } catch (error) {
    if (error !== 'cancel') {
      console.error('Failed to submit:', error)
      ElMessage.error(t('farmerDemand.submitFailed'))
    }
  }
}

// 批量提交审核
const handleBatchSubmit = async () => {
  if (selectedRows.value.length === 0) {
    ElMessage.warning(t('farmerDemand.pleaseSelectData'))
    return
  }

  try {
    await ElMessageBox.confirm(
      t('farmerDemand.batchSubmitConfirm', { count: selectedRows.value.length }),
      t('common.tip'),
      {
        confirmButtonText: t('common.confirm'),
        cancelButtonText: t('common.cancel'),
        type: 'warning'
      }
    )

    const ids = selectedRows.value.map(row => row.id)
    const res = await submitForAudit(ids)

    if (res.code === 200) {
      const result = res.data
      if (result.successCount > 0) {
        ElMessage.success(
          t('farmerDemand.batchSubmitResult', {
            success: result.successCount,
            fail: result.failCount
          })
        )
        selectedRows.value = []
        loadData()
      } else {
        ElMessage.error(t('farmerDemand.submitFailed'))
      }
    } else {
      ElMessage.error(res.msg || t('farmerDemand.submitFailed'))
    }
  } catch (error) {
    if (error !== 'cancel') {
      console.error('Failed to batch submit:', error)
      ElMessage.error(t('farmerDemand.submitFailed'))
    }
  }
}
```

## 🎯 功能说明

### 单条提交审核
- 点击列表中某一行的"提交审核"按钮
- 只有"草稿"和"已驳回"状态的需求可以提交

### 批量提交审核
1. 勾选要提交的需求(只能勾选草稿和驳回状态)
2. 点击顶部的"提交审核 (N)"按钮
3. 确认后批量提交

### 提交后状态变化
- 状态从 `draft` 变为 `submitted`
- 审核层级设为 `village`(村级)
- 生成审核记录

---

## 📱 移动端支持

- 卡片左侧添加复选框
- 可勾选多条进行批量提交
- "提交审核"按钮显示在卡片操作区

---

## ⚠️ 注意事项

1. 确保后端数据库表 `demand_audit_record` 已经按照 `fix_demand_audit_record.sql` 修复
2. 只有"草稿"和"已驳回"状态的需求可以提交审核
3. 提交成功后需求状态会变为"已提交",进入审核流程
4. 审核流程: 村级 → 镇级 → 区级 → 州级 → 部级
